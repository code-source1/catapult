#include <iostream>
#define MAX_POINTS 100
#define HASH_BASE 101

class Strategy {
public:
    virtual int findMaxAligned(int pts[][2], int count) = 0;
    virtual ~Strategy() {}
};

class AlignCalc : public Strategy {
public:
    int findMaxAligned(int pts[][2], int count) override {
        if (count <= 2) return count;
        int maxAligned = 0;
        for (int i = 0; i < count; i++) {
            int hashTable[HASH_BASE] = {0};
            int localMax = 0;
            for (int j = i + 1; j < count; j++) {
                int deltaX = pts[j][0] - pts[i][0];
                int deltaY = pts[j][1] - pts[i][1];
                int index = (deltaX * 31 + deltaY) % HASH_BASE;
                hashTable[index]++;
                localMax = std::max(localMax, hashTable[index]);
            }
            maxAligned = std::max(maxAligned, localMax + 1);
        }
        return maxAligned;
    }
};

class Points {
public:
    int data[MAX_POINTS][2];
    int n;
    Strategy* calculation;

    Points() : calculation(new AlignCalc()) {}

    ~Points() {
        delete calculation;
    }

    int getMaxAligned() {
        return calculation->findMaxAligned(data, n);
    }
};
